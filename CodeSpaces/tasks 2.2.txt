#include <stddef.h>

typedef unsigned int u32;
typedef unsigned short u16;
volatile u16 *video = (volatile u16 *)0xB8000;
static int cursor = 0;

static void putc(char c) {
    if (c == '\n') {
        cursor = (cursor / 80 + 1) * 80;
    } else {
        if (cursor >= 80 * 25) cursor = 0;
        video[cursor++] = (u16)(c | 0x0F00);
    }
}
static void puts(const char *s) { while (*s) putc(*s++); }
static int contains(const char *s, const char *sub) {
    while (*s) {
        const char *a = s;
        const char *b = sub;
        while (*a && *b && *a == *b) { ++a; ++b; }
        if (!*b) return 1;
        ++s;
    }
    return 0;
}
struct multiboot_info { u32 flags, mem_lower, mem_upper, boot_device, cmdline, mods_count, mods_addr, syms[4], mmap_length, mmap_addr, drives_length, drives_addr, config_table, boot_loader_name, apm_table, vbe_control_info, vbe_mode_info; u16 vbe_mode, vbe_interface_seg, vbe_interface_off, vbe_interface_len; };
void kmain(unsigned int magic, unsigned int mbi_addr) {
    (void)magic; (void)mbi_addr;
    const char *cmdline = "";
    if (magic == 0x2BADB002) {
        struct multiboot_info *mbi = (struct multiboot_info *)mbi_addr;
        if (mbi && (mbi->flags & 0x4)) cmdline = (const char *)mbi->cmdline;

    puts("Linux 2.2 demo\n");

    for (;;) __asm__ volatile("hlt");
}
EOF
